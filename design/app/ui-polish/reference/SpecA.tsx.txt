function Cite({ c, open: o = false }: { c: Citation; open?: boolean }) {
  const [open, setOpen] = useState(o);
  const lines = c.excerpt.split("\n");
  const start = c.chunk_index + 1;
  const short = c.document_id.replace("didi-build/", "");
  return (
    <div className="cite">
      <button type="button" className="cite-h" aria-expanded={open} onClick={() => setOpen(!open)}>
        <span style={{ display: "inline-flex", transform: open ? "rotate(90deg)" : "none", transition: "transform .15s" }}><IcChevron size={12} /></span>
        <span className="path">{short}</span>
        <span>L{start}{lines.length > 1 ? `:${start + lines.length - 1}` : ""}</span>
      </button>
      {open ? <pre>{lines.map((l, i) => <div key={i} className={i === 0 ? "hit" : undefined}><span className="ln">{start + i}</span>{l}</div>)}</pre> : null}
    </div>
  );
}

function Seat({ o, compact }: { o: SeatData; compact?: boolean }) {
  const [open, setOpen] = useState(!compact);
  const head = (<><span className="seat-mark" style={{ color: o.color }}>{o.mark}</span><span className="seat-name">{o.name}</span><Verdict v={o.verdict} /></>);
  return (
    <article className="seat" data-compact={compact ? "true" : "false"} data-open={open ? "true" : "false"}>
      {compact ? (
        <button type="button" className="seat-h seat-toggle" aria-expanded={open} onClick={() => setOpen(!open)}>{head}<span className="chev" aria-hidden="true"><IcDown size={14} /></span></button>
      ) : <div className="seat-h">{head}</div>}
      {open ? (<><p>{o.reasoning}</p><div className="seat-cites">{o.citations.map((c) => <Cite key={c.document_id} c={c} />)}</div></>) : null}
    </article>
  );
}

const citeLabel = (c: Citation) => {
  const base = c.document_id.split("/").pop() ?? c.document_id;
  return c.knowledge_source_id === "src-linear" ? c.document_id.replace("Didi Build / ", "") : `${base} L${c.chunk_index + 1}`;
};

/* ---------- write-back ---------- */
export type WB = "idle" | "proposing" | "plan" | "refine" | "refined" | "empty" | "executing" | "created" | "failed";

function WriteBack({ wb }: { wb: WB }) {
  const [feedback, setFeedback] = useState(wb === "refine" ? FEEDBACK_DRAFT : "");
  if (wb === "idle" || wb === "proposing") {
    return (
      <div className="chair-actions">
        <button type="button" className="btn btn-primary btn-sm" disabled={wb === "proposing"}>
          {wb === "proposing" ? <><Spin />Proposing write-back</> : "Propose write-back"}
        </button>
      </div>
    );
  }
  if (wb === "created") {
    return (
      <section className="wb wb-done" aria-label="Write-back created">
        <div className="wb-h"><span className="wb-ok"><IcCheck size={12} /></span><b>Written back</b></div>
        <ul className="wb-results">
          {RESULTS.map((r) => (
            <li key={r.url}>
              {r.kind === "github_pr" ? <IcGitHub size={14} /> : <IcLinear size={13} />}
              <a href={r.url} className="link">{r.label}<IcExternal size={12} /></a>
            </li>
          ))}
        </ul>
      </section>
    );
  }
  const lines = wb === "refined" ? PLAN_LINES_REFINED : PLAN_LINES;
  const busy = wb === "executing";
  return (
    <section className="wb" aria-label="Write-back plan">
      <div className="wb-h"><b>Write-back plan</b><span className="small muted">Nothing is created until you confirm.</span></div>
      {wb === "failed" ? (
        <div className="callout callout-err" role="alert">
          <b>Nothing was created.</b> Something went wrong on our side, so it's safe to try again.
        </div>
      ) : null}
      {wb === "empty" ? (
        <p className="wb-empty small">No follow-up action needed. The recommendation doesn't call for a doc change or a roadmap item.</p>
      ) : (
        <ul className="wb-lines">
          {lines.map((l) => (
            <li key={l}>{l.startsWith("Create Linear") ? <IcLinear size={13} /> : <IcFile size={14} />}<span>{l}</span></li>
          ))}
        </ul>
      )}
      <label className="wb-field">
        <span className="small muted">Suggest a change</span>
        <textarea rows={2} value={feedback} disabled={busy} placeholder="Optional refinement notes" onChange={(e) => setFeedback(e.target.value)} />
      </label>
      <div className="wb-actions">
        <button type="button" className="btn btn-secondary btn-sm" disabled={busy || wb === "refine" || !feedback.trim()}>
          {wb === "refine" ? <><Spin />Refining</> : "Refine"}
        </button>
        {wb === "empty" ? null : wb === "failed" ? (
          <button type="button" className="btn btn-primary btn-sm"><IcRefresh size={13} />Retry</button>
        ) : (
          <button type="button" className="btn btn-primary btn-sm" disabled={busy || wb === "refine"}>{busy ? <><Spin />Creating</> : "Confirm"}</button>
        )}
      </div>
    </section>
  );
}

function Chair({ seats, wb = "idle", innerRef }: { seats: SeatData[]; wb?: WB; innerRef?: Ref<HTMLElement> }) {
  const cites: Citation[] = [];
  for (const s of seats) for (const c of s.citations) if (!cites.some((x) => x.document_id === c.document_id)) cites.push(c);
  const [sel, setSel] = useState<string | null>(null);
  const picked = cites.find((c) => c.document_id === sel);
  return (
    <article className="chair" ref={innerRef} aria-label="Chair's recommendation">
      <div className="chair-h">
        <span className="chair-mark"><DomMark size={18} /></span>
        <span className="ovl">Chair's recommendation</span>
        <Verdict v={COUNCIL.overall_verdict} />
      </div>
      <p className="syn">{COUNCIL.synthesis}</p>
      <div className="chair-cites">
        <span className="ovl ovl-sm">Cited by the council</span>
        <div className="chips">
          {cites.map((c) => (
            <button type="button" key={c.document_id} className="cchip" aria-pressed={sel === c.document_id} title={c.document_id} onClick={() => setSel(sel === c.document_id ? null : c.document_id)}>
              {c.knowledge_source_id === "src-linear" ? <IcLinear size={11} /> : <IcGitHub size={12} />}<span>{citeLabel(c)}</span>
            </button>
          ))}
        </div>
        {picked ? <pre className="chair-pre"><span className="chair-pre-h">{picked.document_id}</span>{"\n"}{picked.excerpt}</pre> : null}
      </div>
      <WriteBack key={wb} wb={wb} />
    </article>
  );
}

/* ---------- chat shell ---------- */
type DockMode = "ready" | "none" | "busy" | "connecting" | "closed" | "error";
function Dock({ mode, mobile }: { mode: DockMode; mobile?: boolean }) {
  const [text, setText] = useState("");
  const disabled = mode === "busy" || mode === "connecting" || mode === "closed";
  return (
    <div className="dock">
      <div className="dock-in">
        {mode === "connecting" ? (
          <div className="strip"><Spin /><span>Connecting live updates</span></div>
        ) : mode === "closed" ? (
          <div className="strip strip-err" role="alert">
            <span>Live updates stopped. Refresh the page to reconnect.</span>
            <button type="button" className="btn btn-secondary btn-sm"><IcRefresh size={13} />Refresh</button>
          </div>
        ) : mode === "error" ? (
          <div className="strip strip-err" role="alert">
            <span>Something went wrong on our side. Try again in a moment.</span>
            <button type="button" className="icon-btn" aria-label="Dismiss"><IcX size={14} /></button>
          </div>
        ) : mode === "none" ? (
          <a className="srcline" href="#connections"><span className="dot dot-idle" />No sources yet. Add one in Connections</a>
        ) : (
          <a className="srcline" href="#connections"><span className="dot dot-ready" />Answering from 2 sources</a>
        )}
        <form className="composer" data-disabled={disabled} onSubmit={(e) => { e.preventDefault(); setText(""); }}>
          <textarea rows={1} value={text} disabled={disabled} onChange={(e) => setText(e.target.value)}
            placeholder={mode === "busy" ? "Quorum is answering" : mobile ? "Ask a question" : "Ask a question or submit a proposal for review"} aria-label="Message" />
          <button type="submit" className="send" aria-label="Send" disabled={disabled || !text.trim()}><IcSend size={16} /></button>
        </form>
      </div>
    </div>
  );
}

function ChatShell({ children, dock, mobile, top = false }: { children: ReactNode; dock: DockMode; mobile?: boolean; top?: boolean }) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    if (top) return;
    const go = () => { const el = ref.current; if (el) el.scrollTop = el.scrollHeight; };
    go(); const t = setTimeout(go, 120); return () => clearTimeout(t);
  }, [top]);
  return (
    <div className="chat-shell">
      <Bar active="chat" />
      <div className="scroller" ref={ref}><section className="thread" aria-live="polite">{children}</section></div>
      <Dock mode={dock} mobile={mobile} />
    </div>
  );
}

const You = ({ text }: { text: string }) => <div className="you">{text}</div>;
const Who = ({ route }: { route: string }) => <div className="who"><DomMark size={18} /><b>Quorum</b><span className="route">{route}</span></div>;

function Chat({ state, mobile }: { state: string; mobile?: boolean }) {
  if (state === "c-nosources") {
    return (
      <ChatShell dock="none" mobile={mobile} top>
        <div className="empty-hero">
          <DomMark size={40} />
          <h2>Connect a source to start</h2>
          <p className="muted">Quorum answers from your code and issues, so it needs something to read first. Most repos are ready in a few minutes.</p>
          <div className="empty-cards">
            <a className="empty-card" href="#connections"><IcGitHub size={18} /><span><b>GitHub</b><span className="small muted">Repositories you choose</span></span><span className="btn btn-secondary btn-sm">Connect</span></a>
            <a className="empty-card" href="#connections"><IcLinear size={16} /><span><b>Linear</b><span className="small muted">Your team's issues</span></span><span className="btn btn-secondary btn-sm">Connect</span></a>
          </div>
          <a className="link small" href="#connections">Go to Connections<IcChevron size={12} /></a>
        </div>
      </ChatShell>
    );
  }
  if (state === "c-history") {
    return (
      <ChatShell dock="ready" mobile={mobile} top>
        <div className="loading-line" role="status"><Spin />Loading chat history</div>
        <div className="skel-thread" aria-hidden="true"><i className="r" /><i /><i /><i className="s" /></div>
      </ChatShell>
    );
  }
  if (state === "c-empty") {
    return (
      <ChatShell dock="ready" mobile={mobile} top>
        <div className="empty-hero">
          <DomMark size={40} />
          <h2>Ask about your code, issues or a decision</h2>
          <p className="muted">Quick questions get a direct answer with sources. Proposals and hard calls go to the council: three seats review it, then the chair recommends.</p>
        </div>
      </ChatShell>
    );
  }
  const connection = state === "c-connecting" ? "connecting" : state === "c-closed" ? "closed" : state === "c-senderror" ? "error" : "ready";
  if (state === "c-nothing") {
    return (
      <ChatShell dock="ready" mobile={mobile}>
        <You text={NOTHING_Q} />
        <Who route="Direct answer" />
        <div className="answer">
          <p>{NOTHING_ANSWER}</p>
          <div className="answer-foot"><a className="btn btn-secondary btn-sm" href="#connections">Open Connections</a></div>
        </div>
      </ChatShell>
    );
  }
  if (state === "c-stub") {
    return (
      <ChatShell dock="ready" mobile={mobile}>
        <You text={STUB_Q} />
        <Who route="Linear request" />
        <div className="note">
          <IcLinear size={16} />
          <div><b>Linear actions from chat aren't available yet</b><p className="small muted">Quorum understood this as a Linear request. For now, ask about your code, issues or a decision.</p></div>
        </div>
      </ChatShell>
    );
  }
  /* direct answer, also used as the backdrop for connection and send-error states */
  return (
    <ChatShell dock={connection as DockMode} mobile={mobile}>
      <You text={DIRECT_Q} />
      <Who route="Direct answer" />
      <div className="answer">
        <p>{DIRECT_ANSWER}</p>
        <div className="answer-cites"><span className="ovl ovl-sm">Sources</span>{DIRECT_CITES.map((c, i) => <Cite key={c.document_id} c={c} open={i === 0} />)}</div>
      </div>
      {state === "c-senderror" ? <You text="And where is the TTL configured?" /> : null}
    </ChatShell>
  );
}

function Council({ state, mobile }: { state: string; mobile?: boolean }) {
  if (state === "k-result" || state === "k-result10" || state.startsWith("w-")) {
    const seats = state === "k-result10" ? SEATS10 : SEATS3;
    const wb: WB = state.startsWith("w-") ? (state.slice(2) === "propose" ? "idle" : (state.slice(2) as WB)) : "idle";
    return (
      <ChatShell dock="ready" mobile={mobile}>
        <You text={QUESTION} />
        <Who route="Council review" />
        <div className="seats" data-count={seats.length}>{seats.map((o) => <Seat key={o.id} o={o} compact={mobile} />)}</div>
        <Chair seats={seats} wb={wb} />
      </ChatShell>
    );
  }
  const map: Record<string, [string, SeatStatus[]]> = {
    "k-alerting": ["alerting_council", [null, null, null]],
    "k-waiting": ["waiting_on_council", ["pondering", "researching", "pondering"]],
    "k-deliberating": ["council_deliberating", ["recommendation_ready", "giving_recommendation", "recommendation_ready"]],
  };
  return (
    <ChatShell dock="busy" mobile={mobile}>
      <You text={QUESTION} />
      {state === "k-understanding" ? (
        <><Who route="" /><div className="understanding" role="status"><Pulse />Understanding the message</div></>
      ) : (
        <><Who route="Council review" /><PendingCouncil sup={map[state][0]} statuses={map[state][1]} mobile={mobile} /></>
      )}
    </ChatShell>
  );
}

/* ---------- connections ---------- */
const statusText = (s: KnowledgeSource) =>
  s.status === "ready" ? "Ready" : s.status === "indexing" ? "Indexing. Usually a few minutes." : s.status === "pending" ? "Waiting to start" : "Can't read this source";

function SourceRow({ s, menu = false, confirm = false, refreshing = false, inline = false }: { s: KnowledgeSource; menu?: boolean; confirm?: boolean; refreshing?: boolean; inline?: boolean }) {
  const [m, setM] = useState(menu);
  const [c, setC] = useState(confirm);
  return (
    <div className="row" data-confirm={c}>
      <SourceIcon type={s.source_type} className="row-ic" />
      <div className="row-b">
        <span className="row-name">{s.name}</span>
        {refreshing ? <span className="row-st"><Spin />Refreshing</span> : <span className={`row-st ${s.status}`}><span className={`dot dot-${s.status}`} />{statusText(s)}</span>}
        {s.status === "error" && s.status_message ? (
          <div className="row-err">
            <span>{s.status_message}</span>
            <span className="row-err-actions">
              <a className="btn btn-secondary btn-sm" href="#gh">Check GitHub access <IcExternal size={12} /></a>
              <button type="button" className="btn btn-quiet btn-sm"><IcRefresh size={13} />Refresh</button>
            </span>
          </div>
        ) : null}
        {inline && m ? (
          <div className="row-actions" role="menu" aria-label={`Actions for ${s.name}`}>
            <button type="button" role="menuitem" onClick={() => setM(false)}><IcRefresh size={14} /><span><b>Refresh</b><span className="small muted">Read the latest changes</span></span></button>
            <button type="button" role="menuitem" className="danger" onClick={() => { setM(false); setC(true); }}><IcTrash size={14} /><span><b>Remove</b><span className="small muted">Stop using it in answers</span></span></button>
          </div>
        ) : null}
        {c ? (
          <div className="confirm" role="alertdialog" aria-label={`Remove ${s.name}`}>
            <span><b>Remove {s.name}?</b> Quorum stops answering from it. You can add it again any time.</span>
            <span className="confirm-actions">

            {lockGroup ? <span className="pt-hint">{grp.label}</span> : (
              <div className="pt-grp" role="group" aria-label="Screen">
                {GROUPS.map((x) => <button key={x.id} type="button" aria-pressed={group === x.id} onClick={() => { setGroup(x.id); setState(x.states[0].id); }}>{x.label}</button>)}
              </div>
            )}
            <div className="pt-grp" role="group" aria-label="Theme">
              <button type="button" aria-pressed={theme === "dark"} onClick={() => setTheme("dark")}>Dark</button>
              <button type="button" aria-pressed={theme === "light"} onClick={() => setTheme("light")}>Light</button>
            </div>
            <div className="pt-grp" role="group" aria-label="Device">
              <button type="button" aria-pressed={!mobile} onClick={() => setDevice("desktop")}>Desktop</button>
              <button type="button" aria-pressed={mobile} onClick={() => setDevice("mobile")}>380px</button>
            </div>
            <button type="button" className="pt-x" aria-label="Hide controls" onClick={() => setOpen(false)}>Hide</button>
          </div>
          <div className="pt-row">
            <div className="pt-grp pt-states" role="group" aria-label="State">
              {grp.states.map((x) => <button key={x.id} type="button" aria-pressed={state === x.id} onClick={() => setState(x.id)}>{x.label}</button>)}
            </div>
          </div>
        </div>
      ) : full ? null : <button type="button" className="pt-open" onClick={() => setOpen(true)}>Review controls</button>}
      <div className="pt-stage">
        <Screen key={`${state}-${device}-${theme}`} state={state} theme={theme} mobile={mobile} full={full} />
      </div>
    </div>
  );
}
