/* Picker state shared by every variant: search, select, already-added rows disabled. */
import { useMemo, useState } from "react";
export function usePicker(preselected: string[] = ["r4", "t2"]) {
  const [tab, setTab] = useState<SourceType>("github_repo");
  const [query, setQuery] = useState("");
  const [selected, setSelected] = useState<string[]>(preselected);
  const [added, setAdded] = useState<number | null>(null);
  const list = tab === "github_repo" ? REPOS : TEAMS;
  const filtered = useMemo(
    () => list.filter((c) => c.name.toLowerCase().includes(query.trim().toLowerCase())),
    [list, query],
  );
  const toggle = (c: Candidate) => {
    if (c.added) return;
    setSelected((s) => (s.includes(c.id) ? s.filter((x) => x !== c.id) : [...s, c.id]));
  };
  const count = selected.length;
  const countIn = (k: SourceType) => selected.filter((id) => (k === "github_repo" ? REPOS : TEAMS).some((c) => c.id === id)).length;
  const confirm = () => {
    if (!count) return;
    setAdded(count);
    setSelected([]);
    window.setTimeout(() => setAdded(null), 2600);
  };
  return { tab, setTab, query, setQuery, selected, toggle, filtered, count, countIn, confirm, added, list };
}
export type Picker = ReturnType<typeof usePicker>;
